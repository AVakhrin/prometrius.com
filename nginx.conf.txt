# Основной домен. На время получения сертификата отдаём сайт по HTTP —
# после certbot здесь появится перенаправление на HTTPS.
server {
    server_name prometrius.com www.prometrius.com;

    root /var/www/prometrius.com;
    index index.html;
    client_max_body_size 16M;

    # Проверка сертификата должна проходить без перенаправлений,
    # иначе certbot не сможет подтвердить домен
    location ^~ /.well-known/acme-challenge/ {
        allow all;
    }

    location / {
        try_files $uri $uri/ =404;
    }

    location /forum/ {
        index index.php index.html;
        try_files $uri $uri/ /forum/index.php?$query_string;
    }

    location ~ ^/forum/.+\.php$ {
        fastcgi_split_path_info ^(.+\.php)(/.+)$;
        fastcgi_pass unix:/run/php/php8.1-fpm.sock;
        fastcgi_index index.php;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param PATH_INFO $fastcgi_path_info;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/prometrius.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/prometrius.com/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
